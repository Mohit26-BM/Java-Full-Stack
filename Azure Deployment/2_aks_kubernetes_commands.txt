# AZURE KUBERNETES SERVICE (AKS) - deploy KubernetesDemoApp, 2 pods behind a load balancer
# Use Windows PowerShell (prompt starts with PS), not Command Prompt. Docker Desktop must be running.
# Paste one block at a time in the SAME window and check the "expect" line first.
# COST: about $4.5/day (2 VMs, disks, load balancer, IPs, registry). Delete the same day (3_cleanup_commands.txt).
# Region: centralindia blocks every normal VM size for this subscription, so use indiasouthcentral
# (Standard_B2s is open there; quota is 4 vCPUs for the BS family = exactly 2 nodes).

# ---- Block 1: variables ----
az account show --output table

$LOCATION = "indiasouthcentral"
$RG = "Mohit-rg"
$ACR = "mohitdemoacr" + (Get-Random -Maximum 99999)
$IMAGE = "kubernetesdemoapp"
$TAG = "v3"
$AKS = "mohit-aks-cluster"
$VMSIZE = "Standard_B2s"
$APPDIR = "D:\CSB_Training\Kubernets\KubernetesDemoApp"
$YAML = "D:\CSB_Training\Azure Deployment\k8s-demo-aks.yaml"
$ACR

# ---- Block 2: register the providers AKS needs (first time only) ----
az provider register --namespace Microsoft.ContainerService
az provider register --namespace Microsoft.Compute
az provider register --namespace Microsoft.Network

az provider show --namespace Microsoft.ContainerService --query registrationState --output tsv
az provider show --namespace Microsoft.Compute --query registrationState --output tsv
az provider show --namespace Microsoft.Network --query registrationState --output tsv
# expect: Registered x3. Repeat the show lines every minute until they are.

# ---- Block 3: check the VM size is allowed and there is quota (can be slow, Ctrl+C to skip) ----
az vm list-usage --location $LOCATION --query "[?name.value=='cores' || name.value=='standardBSFamily'].{name:name.localizedValue,used:currentValue,limit:limit}" --output table
# expect: Total Regional vCPUs and Standard BS Family with room for 4

# ---- Block 4: create the registry ----
az acr create --resource-group $RG --name $ACR --location $LOCATION --sku Basic --admin-enabled false
$LOGIN_SERVER = az acr show --name $ACR --query loginServer --output tsv
$FULL_IMAGE = "${LOGIN_SERVER}/${IMAGE}:${TAG}"
$FULL_IMAGE
# expect: mohitdemoacrNNNNN.azurecr.io/kubernetesdemoapp:v3

# ---- Block 5: build and push the image ----
Push-Location $APPDIR
mvn -q clean package -DskipTests
docker build -t "${IMAGE}:${TAG}" .
Pop-Location

docker tag "${IMAGE}:${TAG}" $FULL_IMAGE
az acr login --name $ACR
docker push $FULL_IMAGE
az acr repository show-tags --name $ACR --repository $IMAGE --output table
# expect: your tag listed

# ---- Block 6: create the cluster (5 to 10 minutes). --attach-acr lets the nodes pull from the registry ----
az aks create `
  --resource-group $RG `
  --name $AKS `
  --location $LOCATION `
  --node-count 2 `
  --node-vm-size $VMSIZE `
  --generate-ssh-keys `
  --attach-acr $ACR `
  --tier free `
  --output table
# if it fails on size or quota, paste the error, do not retry the same command

# ---- Block 7: connect kubectl to the cluster ----
az aks get-credentials --resource-group $RG --name $AKS --overwrite-existing
kubectl config current-context
kubectl get nodes -o wide
# expect: 2 nodes, STATUS Ready. kubectl now points at AKS, not docker-desktop

# ---- Block 8: write the Kubernetes YAML (Deployment with 2 pods + LoadBalancer Service) ----
@"
apiVersion: apps/v1
kind: Deployment
metadata:
  name: k8s-demo
spec:
  replicas: 2
  selector:
    matchLabels:
      app: k8s-demo
  template:
    metadata:
      labels:
        app: k8s-demo
    spec:
      topologySpreadConstraints:
        - maxSkew: 1
          topologyKey: kubernetes.io/hostname
          whenUnsatisfiable: ScheduleAnyway
          labelSelector:
            matchLabels:
              app: k8s-demo
      containers:
        - name: k8s-demo
          image: $FULL_IMAGE
          ports:
            - containerPort: 8081
          resources:
            requests:
              cpu: 100m
              memory: 256Mi
            limits:
              cpu: 500m
              memory: 512Mi
---
apiVersion: v1
kind: Service
metadata:
  name: k8s-demo-lb
spec:
  type: LoadBalancer
  selector:
    app: k8s-demo
  ports:
    - port: 80
      targetPort: 8081
"@ | Out-File -FilePath $YAML -Encoding ascii
Get-Content $YAML | Select-String image:
# expect: your real registry name in the image line

# ---- Block 9: deploy ----
kubectl apply -f $YAML
kubectl rollout status deployment/k8s-demo --timeout=5m
kubectl get pods -o wide
# expect: 2 pods Running

# ---- Block 10: public IP (a few minutes) and test ----
kubectl get service k8s-demo-lb --watch
# press Ctrl+C when EXTERNAL-IP shows an address

$IP = kubectl get service k8s-demo-lb -o jsonpath='{.status.loadBalancer.ingress[0].ip}'
Start-Process "http://$IP/"
1..10 | ForEach-Object { curl.exe -s "http://$IP/api/info" }
# expect: the page, and both pod names across the 10 requests
# (a browser reuses one connection so it may stay on one pod; separate curl calls alternate)
