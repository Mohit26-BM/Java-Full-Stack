az identity delete --name mohit-demo-identity --resource-group Mohit-rg

az monitor log-analytics workspace delete --workspace-name <workspace name from the list> --resource-group Mohit-rg --yes

az acr list --resource-group Mohit-rg --query "[].name" --output tsv

az resource list --resource-group Mohit-rg --output table
az group list --output table
