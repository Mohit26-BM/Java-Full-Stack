    $LOCATION = "centralindia"
    $RG = "Mohit-rg"
    $ACR = "mohitdemoacr" + (Get-Random -Maximum 99999)
    $IMAGE = "kubernetesdemoapp"
    $TAG = "v1"
    $APP = "mohit-demo-app"
    $ENVNAME = "mohit-demo-env"
    $IDENTITY = "mohit-demo-identity"
    $APPDIR = "D:\CSB_Training\Kubernets\KubernetesDemoApp"
    $TARGET_PORT = "8081"

    az extension add --name containerapp --upgrade
    az provider register --namespace Microsoft.App --wait
    az provider register --namespace Microsoft.OperationalInsights --wait
    az provider register --namespace Microsoft.ContainerRegistry --wait

    az acr create --resource-group $RG --name $ACR --location $LOCATION --sku Basic --admin-enabled false
    $LOGIN_SERVER = az acr show --name $ACR --query loginServer --output tsv
    $FULL_IMAGE = "${LOGIN_SERVER}/${IMAGE}:${TAG}"

    Push-Location $APPDIR
    .\mvnw.cmd -q clean package -DskipTests
    docker build -t "${IMAGE}:${TAG}" .
    Pop-Location

    docker run -d --name demo-local -p 8082:8081 "${IMAGE}:${TAG}"
    Start-Process http://localhost:8082/hello
    docker stop demo-local
    docker rm demo-local

    docker tag "${IMAGE}:${TAG}" $FULL_IMAGE
    az acr login --name $ACR
    docker push $FULL_IMAGE
    az acr repository show-tags --name $ACR --repository $IMAGE --output table

    az acr config authentication-as-arm update --registry $ACR --status enabled
    az identity create --name $IDENTITY --resource-group $RG --location $LOCATION
    $IDENTITY_ID = az identity show --name $IDENTITY --resource-group $RG --query id --output tsv
    $PRINCIPAL_ID = az identity show --name $IDENTITY --resource-group $RG --query principalId --output tsv
    $ACR_ID = az acr show --name $ACR --resource-group $RG --query id --output tsv
    az role assignment create --assignee-object-id $PRINCIPAL_ID --assignee-principal-type ServicePrincipal --scope $ACR_ID --role AcrPull
    Start-Sleep -Seconds 60

    az containerapp env create --name $ENVNAME --resource-group $RG --location $LOCATION
    az containerapp create `
      --name $APP `
      --resource-group $RG `
      --environment $ENVNAME `
      --image $FULL_IMAGE `
      --target-port $TARGET_PORT `
      --ingress external `
      --user-assigned $IDENTITY_ID `
      --registry-identity $IDENTITY_ID `
      --registry-server $LOGIN_SERVER

    $FQDN = az containerapp show --name $APP --resource-group $RG --query properties.configuration.ingress.fqdn --output tsv
    $URL = "https://$FQDN"
    $URL
    Invoke-WebRequest "$URL/hello" -UseBasicParsing | Select-Object StatusCode, Content

    az containerapp logs show --name $APP --resource-group $RG --type console --tail 100
    az containerapp logs show --name $APP --resource-group $RG --type system --tail 100

    az containerapp delete --name $APP --resource-group $RG --yes
    az containerapp env delete --name $ENVNAME --resource-group $RG --yes
    az acr delete --name $ACR --resource-group $RG --yes
    az identity delete --name $IDENTITY --resource-group $RG
